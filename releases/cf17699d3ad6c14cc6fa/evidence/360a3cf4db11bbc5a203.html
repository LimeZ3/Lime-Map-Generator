def pixel_objective(error, clean, mode="baseline"):
    if mode == "baseline":
        weights = torch.ones_like(clean)
        weights[:, 2:10] = 1 + 12 * ((clean[:, 2:10] + 1) * .5).clamp(0, 1)
        return (error * weights).mean((2, 3)) / weights.mean((2, 3))
    if mode != "balanced":
        raise ValueError("Unknown loss")
    ordinary = error.mean((2, 3))
    # Continuous source activity, not an analytical input or an eligibility mask.
    activity = ((clean[:, 2:10] + 1) * .5).clamp(0, 1)
    total = activity.sum((2, 3))
    foreground = (error[:, 2:10] * activity).sum((2, 3)) / total.clamp_min(1e-12)
    sparse = torch.where(total > 1e-12,
                         .5 * ordinary[:, 2:10] + .5 * foreground, ordinary[:, 2:10])
    return torch.cat([ordinary[:, :2], sparse, ordinary[:, 10:]], dim=1)


def diagnostic_loss(model, clean, ratings, cumulative, generator, mode="baseline", timestep=None):
    device = clean.device
    t = (torch.randint(len(cumulative), (len(clean),), device=device, generator=generator)
         if timestep is None else torch.full((len(clean),), timestep, device=device, dtype=torch.long))
    alpha = cumulative[t, None, None, None]
    noise = torch.randn(clean.shape, device=device, generator=generator)
    noisy = alpha.sqrt() * clean + (1-alpha).sqrt() * noise
    velocity = alpha.sqrt() * noise - (1-alpha).sqrt() * clean
    predicted = model(noisy, t, ratings).float()
    per_channel = pixel_objective((predicted-velocity).square(), clean, mode)
    estimate = alpha.sqrt() * noisy - (1-alpha).sqrt() * predicted
    return per_channel, estimate


def ddim_update(noisy, velocity, alpha, previous, lower=None, upper=None, clipping="legacy"):
    raw = alpha.sqrt() * noisy - (1-alpha).sqrt() * velocity
    clean = raw if clipping == "none" else raw.maximum(lower).minimum(upper)
    epsilon = alpha.sqrt() * velocity + (1-alpha).sqrt() * noisy
    if clipping == "recomputed":
        epsilon = (noisy-alpha.sqrt()*clean)/(1-alpha).sqrt().clamp_min(1e-12)
    return previous.sqrt()*clean+(1-previous).sqrt()*epsilon, raw


@torch.no_grad()
def diagnostic_sample(model, profile, norm, seed=303, steps=48, guidance=1.5, clipping="legacy", size=256):
    model.eval()
    device = next(model.parameters()).device
    noise = np.random.default_rng(seed).standard_normal((1, 13, size, size), dtype=np.float32)
    x = torch.from_numpy(noise).to(device)
    cumulative = schedule(1000, device)
    times = np.linspace(999, 0, steps, dtype=int)
    lower, upper = [np.asarray(norm[k], np.float32).copy() for k in ("sampling_min", "sampling_max")]
    span = np.maximum(upper-lower, .1)
    lower[0] -= .2*span[0]
    upper[:10] += .2*span[:10]
    lo, hi = [torch.as_tensor(a, device=device)[None, :, None, None] for a in (lower, upper)]
    for i, step in enumerate(times):
        t = torch.full((1,), int(step), device=device, dtype=torch.long)
        with torch.autocast(device.type, dtype=torch.bfloat16, enabled=device.type == "cuda"):
            velocity = model(x, t, profile).float()
            if guidance != 1:
                baseline = model(x, t, None).float()
                velocity = baseline + guidance*(velocity-baseline)
        previous = cumulative[times[i+1]] if i+1 < steps else torch.ones((), device=device)
        x, raw = ddim_update(x, velocity, cumulative[step], previous, lo, hi, clipping)
    return raw[0].cpu().numpy(), x[0].cpu().numpy()


def get_data(root, split, keys=None):
    if keys is None:
        return JointDataset(root, split)
    root = Path(root)
    verify_release(root)
    available = {r["key"]:r for r in read_rows(root/(split+".jsonl"))}
    data = JointDataset.__new__(JointDataset)
    data.root, data.norm = root, read_json(root/"normalization.json")
    data.rows = [available[k] for k in keys]
    targets, profiles = [], []
    for row in data.rows:
        file = root/row["archive"]
        if file_hash(file) != row["sha256"]:
            raise ValueError("Changed diagnostic source")
        with np.load(file, allow_pickle=False) as a:
            targets.append(a["target"].astype(np.float16)); profiles.append(a["ratings"])
    data.targets, data.profiles = np.stack(targets), np.stack(profiles)
    return data


def sample_metrics(encoded, norm, reference=None):
    physical = decode(encoded, norm)
    metrics = dict(density_integrals={CHANNELS[i]:float(physical[i].sum()) if np.isfinite(physical[i]).all() else None for i in range(2, 10)},
                   negative_quantity_fraction=float((physical[1:10] < -1e-5).mean()),
                   median_depth=float(np.median(physical[1])))
    if reference is not None:
        metrics["encoded_mse"] = dict(zip(CHANNELS, ((encoded-reference)**2).mean((1,2)).tolist(), strict=True))
        for i, name in ((0, "height"), (10, "red"), (11, "green"), (12, "blue")):
            a, b = encoded[i].ravel(), reference[i].ravel()
            metrics[name+"_correlation"] = float(np.corrcoef(a, b)[0,1]) if min(a.std(), b.std())>1e-8 else None
        target_physical = decode(reference, norm)
        metrics["source_integrals"] = {CHANNELS[i]:float(target_physical[i].sum()) for i in range(2,10)}
        for i,name in ((2,"start_a"),(3,"start_b")):
            predicted=np.asarray(np.unravel_index(np.argmax(encoded[i]),encoded[i].shape))
            actual=np.asarray(np.unravel_index(np.argmax(reference[i]),reference[i].shape))
            metrics[name+"_peak_distance_world"]=float(np.linalg.norm(predicted-actual)*512/encoded.shape[-1])
        activity=np.clip((reference[2:10]+1)*.5,0,1)
        error=(encoded[2:10]-reference[2:10])**2
        metrics["foreground_mse"]={CHANNELS[i+2]:float((error[i]*activity[i]).sum()/activity[i].sum())
                                    if activity[i].sum()>0 else None for i in range(8)}
    return metrics


@torch.no_grad()
def multi_noise_validation(model, data, cumulative, batch=6, timesteps=(100,500,900,990)):
    model.eval()
    results = {}
    for timestep in timesteps:
        generator = torch.Generator(device="cuda").manual_seed(61025)
        losses, reconstruction = [], []
        for start in range(0, len(data.rows), batch):
            clean, ratings = data.batch(range(start,min(start+batch,len(data.rows))), "cuda")
            with torch.autocast("cuda", dtype=torch.bfloat16):
                loss, estimate = diagnostic_loss(model,clean,ratings,cumulative,generator,timestep=timestep)
            losses.extend(loss.cpu().tolist())
            reconstruction.extend((estimate-clean).square().mean((2,3)).cpu().tolist())
        mean = np.mean(losses,axis=0)
        results[str(timestep)] = dict(loss=float(mean.mean()),
            channel_loss=dict(zip(CHANNELS,mean.tolist(),strict=True)),
            clean_mse=dict(zip(CHANNELS,np.mean(reconstruction,axis=0).tolist(),strict=True)))
    return results


def setup(seed):
    if not torch.cuda.is_available():
        raise RuntimeError("CUDA required")
    torch.set_num_threads(4)
    torch.manual_seed(seed)
    torch.backends.cudnn.benchmark = False
    torch.set_float32_matmul_precision("high")


def sampling_probe(args):
    setup(args.seed)
    out = args.out; out.mkdir(parents=True,exist_ok=True)
    saved = torch.load(args.parent,map_location="cpu",weights_only=False,mmap=True)
    config, norm, step = saved["config"],saved["normalization"],saved["step"]
    model = JointUNet(config["base"]).cuda().eval()
    model.load_state_dict(saved["ema"])
    parent_hash = file_hash(args.parent)
    profile = torch.tensor([[0,0,0,0,0,1,0,0,0,.5,0,0]],device="cuda",dtype=torch.float32)
    rows = []
    variants = [(48,1.5,"legacy"),(128,1.5,"legacy"),(128,1.,"legacy"),
                (128,1.5,"none"),(128,1.5,"recomputed")]
    for steps,guidance,clipping in variants:
        name = f"ema_steps{steps}_cfg{guidance}_{clipping}"
        raw, bounded = diagnostic_sample(model,profile,norm,steps=steps,guidance=guidance,clipping=clipping)
        np.savez_compressed(out/(name+".npz"),raw=raw,bounded=bounded,physical=decode(raw,norm))
        layer_strip(raw,norm,name).save(out/(name+".png"))
        rows.append(dict(name=name,steps=steps,guidance=guidance,clipping=clipping,**sample_metrics(raw,norm)))
        print('sampler_variant='+name,flush=True)
    model.load_state_dict(saved["model"])
    raw,bounded = diagnostic_sample(model,profile,norm)
    layer_strip(raw,norm,"RAW model weights, 48 steps CFG1.5").save(out/"raw_weights.png")
    rows.append(dict(name="raw_model_weights",**sample_metrics(raw,norm)))
    del saved; gc.collect()
    model.cpu(); del model; torch.cuda.empty_cache()
    # Reload EMA in a clean allocation for cross-noise evaluation.
    saved=torch.load(args.parent,map_location="cpu",mmap=True,weights_only=False)
    model=JointUNet(config["base"]).cuda().eval();model.load_state_dict(saved["ema"])
    del saved;gc.collect()
    data=get_data(args.dataset,"validation")
    validation=multi_noise_validation(model,data,schedule(1000,"cuda"))
    write_json(out/"summary.json",dict(parent=str(args.parent),parent_sha256=parent_hash,step=step,
        variants=rows,validation=validation,training_changes=False,
        caution="Sampling probes diagnose a frozen checkpoint; no source map is provided to fresh generation"))


def train_probe(args):
    setup(args.seed)
    out=args.out.resolve();out.mkdir(parents=True,exist_ok=True)
    with lease(out/"training.lock"):
        data=get_data(args.dataset,"train",[args.map] if args.map else None)
        validation=data if args.map else get_data(args.dataset,"validation")
        model=JointUNet(base=128,dropout=0. if args.map else .1).cuda()
        ema=copy.deepcopy(model).eval().requires_grad_(False)
        optimizer=torch.optim.AdamW(model.parameters(),lr=args.learning_rate,weight_decay=0. if args.map else .01)
        step,history=0,[]
        lineage=None
        if args.parent:
            saved=torch.load(args.parent,map_location="cuda",weights_only=False)
            model.load_state_dict(saved["model"]);ema.load_state_dict(saved["ema"])
            optimizer.load_state_dict(saved["optimizer"])
            step=saved["step"]
            torch.set_rng_state(saved["torch_rng"].cpu());torch.cuda.set_rng_state(saved["cuda_rng"].cpu())
            lineage=dict(parent=str(args.parent),sha256=file_hash(args.parent),step=step)
            del saved;gc.collect();torch.cuda.empty_cache()
        config=dict(schema="joint_source_diagnostic_v1",dataset=str(args.dataset.resolve()),
            release_sha256=file_hash(args.dataset/"release_manifest.json"),target_channels=CHANNELS,
            spatial_channels=0,profile_dim=12,base=128,parameters=sum(p.numel() for p in model.parameters()),
            train_maps=len(data.rows),validation_maps=len(validation.rows),learning_rate=args.learning_rate,
            loss=args.loss,micro_batch=6,effective_batch=args.batch_size,map=args.map,seed=args.seed,
            fresh_initialization=not bool(args.parent),lineage=lineage,
            augmentation="disabled for memorization" if args.map else "same aligned rotations/reflections/start swap",
            validation_meaning="training-map denoising, not held-out" if args.map else "69 held-out source families",
            code={str(p):file_hash(p) for p in [Path(__file__),Path(__file__).with_name("joint_diffusion.py"),Path(__file__).with_name("multiscale_terrain.py")]},
            test_used=False)
        write_json(out/"training_receipt.json",dict(config=config,supervised_targets=CHANNELS,ratings=RATINGS,
            generation_inputs="13-channel noise, timestep, twelve ratings; no source spatial inputs",
            masks="No analytical/input masks; balanced arm uses continuous source activity for the objective only"))
        write_json(out/"config.json",config)
        snapshot=out/"code_snapshot";snapshot.mkdir(exist_ok=True)
        for path in config["code"]:shutil.copy2(path,snapshot/Path(path).name)
        cumulative=schedule(1000,"cuda")
        started=time.monotonic();start_step=step
        recent=deque(maxlen=100);gradients=deque(maxlen=100);training_stats=[]
        last_eval,last_sample=0.,0.
        best=float("inf")

        def evaluate():
            return multi_noise_validation(ema,validation,cumulative,timesteps=(100,500,900))

        def save(metrics=None):
            nonlocal best
            if metrics:
                history.append(dict(step=step,session_seconds=time.monotonic()-started,metrics=metrics))
                write_json(out/"history.json",history)
            atomic_checkpoint(out/"last.pt",dict(model=model.state_dict(),ema=ema.state_dict(),
                optimizer=optimizer.state_dict(),config=config,normalization=data.norm,step=step,history=history,
                torch_rng=torch.get_rng_state(),cuda_rng=torch.cuda.get_rng_state()))
            if metrics and metrics["500"]["loss"]<best:
                best=metrics["500"]["loss"]
                atomic_checkpoint(out/"best.pt",dict(ema=ema.state_dict(),config=config,normalization=data.norm,step=step))

        def render():
            folder=out/"samples"/f"step_{step:07d}";folder.mkdir(parents=True,exist_ok=True)
            profiles=[(data.rows[0]["key"],data.profiles[0])] if args.map else [("typical",np.median(data.profiles,axis=0))]
            metrics=[]
            for name,profile in profiles:
                for seed in (303,404):
                    raw,bounded=diagnostic_sample(ema,torch.tensor(profile[None],device="cuda"),data.norm,seed=seed)
                    np.savez_compressed(folder/(f"seed{seed}.npz"),raw=raw,bounded=bounded,physical=decode(raw,data.norm))
                    strips=[layer_strip(data.targets[0].astype(np.float32),data.norm,"SOURCE "+data.rows[0]["key"])] if args.map else []
                    strips.append(layer_strip(raw,data.norm,f"FRESH seed {seed}, step {step}, {name}"))
                    canvas=Image.new("RGB",(strips[0].width,sum(s.height for s in strips)))
                    for i,s in enumerate(strips):canvas.paste(s,(0,i*s.height))
                    canvas.save(folder/(f"seed{seed}.png"))
                    if seed==303:canvas.save(out/"latest_layers.png")
                    reference=data.targets[0].astype(np.float32) if args.map else None
                    metrics.append(dict(seed=seed,**sample_metrics(raw,data.norm,reference)))
            write_json(folder/"metrics.json",metrics)
            write_json(out/"latest_samples.json",dict(step=step,folder=str(folder),samples=metrics))

        def progress(state):
            write_json(out/"progress.json",dict(status=state,step=step,started_step=start_step,
                session_seconds=time.monotonic()-started,session_budget=args.seconds,
                mean_training_loss=float(np.mean(recent)) if recent else None,
                gradient_norm_mean=float(np.mean(gradients)) if gradients else None,
                gradient_norm_max=float(max(gradients)) if gradients else None,
                best_validation_500=best if best<float("inf") else None,updated_unix=time.time(),
                loss=args.loss,learning_rate=args.learning_rate,training_maps=len(data.rows),
                peak_allocated_mb=torch.cuda.max_memory_allocated()/1024**2))

        initial=evaluate();save(initial);write_json(out/"initial.json",initial)
        print(json.dumps(dict(initial=initial,config=config)),flush=True)
        while time.monotonic()-started<args.seconds and time.time()<args.deadline-90:
            if (out/"pause.request").exists():break
            step+=1
            rng=np.random.default_rng(np.random.SeedSequence([args.seed,step]))
            generator=torch.Generator(device="cuda").manual_seed(args.seed*1000003+step)
            indices=rng.integers(len(data.rows),size=args.batch_size)
            optimizer.param_groups[0]["lr"]=args.learning_rate*(min(1.,step/50) if args.map else 1.)
            model.train();optimizer.zero_grad(set_to_none=True);total_loss=0.
            for offset in range(0,args.batch_size,6):
                clean,ratings=data.batch(indices[offset:offset+6],"cuda",None if args.map else rng)
                with torch.autocast("cuda",dtype=torch.bfloat16):
                    losses,_=diagnostic_loss(model,clean,ratings,cumulative,generator,args.loss)
                    loss=losses.mean()*len(clean)/args.batch_size
                if not torch.isfinite(loss):raise ValueError("Nonfinite diagnostic loss")
                loss.backward();total_loss+=float(loss.detach())
            grad=nn.utils.clip_grad_norm_(model.parameters(),1.)
            if not torch.isfinite(grad):raise ValueError("Nonfinite diagnostic gradient")
            optimizer.step()
            with torch.no_grad():
                decay=min(.999,(1+step)/(10+step))
                for a,b in zip(ema.parameters(),model.parameters(),strict=True):a.lerp_(b.detach(),1-decay)
            recent.append(total_loss);gradients.append(float(grad))
            if step%20==0:progress("training")
            if step%100==0:
                statistics=dict(step=step,session_seconds=time.monotonic()-started,
                    loss=float(np.mean(recent)),gradient_norm_mean=float(np.mean(gradients)),
                    gradient_norm_max=float(max(gradients)),output_weight_norm=float(model.output[-1].weight.detach().norm()))
                training_stats.append(statistics);write_json(out/"training_stats.json",training_stats)
                print(f'step={step} loss={np.mean(recent):.5f} grad={np.mean(gradients):.3f}',flush=True)
            elapsed=time.monotonic()-started
            if elapsed-last_eval>=300:
                progress("evaluating");save(evaluate());last_eval=time.monotonic()-started
            if elapsed-last_sample >= (300 if args.map else 900):
                progress("sampling");render();last_sample=time.monotonic()-started
        progress("finalizing");final=evaluate();save(final);render();progress("completed")
        write_json(out/"summary.json",dict(step=step,start_step=start_step,initial=initial,final=final,
            session_seconds=time.monotonic()-started,config=config))
