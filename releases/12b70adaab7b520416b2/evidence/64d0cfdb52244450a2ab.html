"""Source-only full-map joint targets; no analysis layers or model execution."""

from __future__ import annotations

import argparse
from collections import Counter
import gzip
import json
from pathlib import Path
import time

import numpy as np

from supmapai.strategy.pipeline_contract import lease
from supmapai.tensors.strategy_dataset import file_hash, point_field, pool, read_json, read_rows
from supmapai.tools.windows_io import write_json

SCHEMA = "joint_source_layers_v1"
CHANNELS = ["height", "water_depth", "start_a", "start_b", "mex", "hydro",
            "natural_energy", "natural_mass", "wreck_mass", "civilians", "red", "green", "blue"]
RATINGS = ["expansion_drop_rating", "offensive_drop_rating", "air_raid_rating",
           "drop_impact_rating", "air_mobility_relevance", "land_attack_relevance",
           "naval_attack_relevance", "mex_supply_rating", "reclaim_abundance_rating",
           "early_economy_rating", "front_openness_rating", "army_control_demand_rating"]
NATURAL = {"tree", "rock", "boulder", "reclaim"}
GRID = 256


def joined_rows(folder):
    return {r["key"]: r for s in ("train", "validation") for r in read_rows(folder/(s+".jsonl"))}


def verify(path, expected):
    if file_hash(path) != expected:
        raise ValueError("Changed source: " + str(path))


def verify_release(folder):
    release = read_json(folder/"release_manifest.json")
    if release["status"] != "ready":
        raise ValueError("Unpublished source: " + str(folder))
    for name, expected in release["artifacts"].items():
        verify(folder/name, expected)
    return file_hash(folder/"release_manifest.json")


def entity_fields(entities, grid=GRID):
    """Absolute per-cell values and source civilian counts, conserving in-canvas sums."""
    from scipy.ndimage import gaussian_filter
    fields = np.zeros((4, grid, grid), np.float64)
    unknown, outside = Counter(), Counter()
    for e in entities:
        if e.get("active_in_configuration") is False:
            continue
        kind = e.get("kind")
        civilian = kind in ("civilian_unit", "civilian_building")
        natural = kind in NATURAL and e.get("source_type") == "prop"
        wreck = kind == "wreckage"
        if not (civilian or natural or wreck):
            continue
        x, z = float(e["x"]), float(e["z"])
        if not (np.isfinite(x) and np.isfinite(z) and 0 <= x < 512 and 0 <= z < 512):
            outside[kind] += 1
            continue
        ix, iz = int(x*grid/512), int(z*grid/512)
        if civilian:
            fields[3, iz, ix] += 1
        for index, name in ([(0, "energy"), (1, "mass")] if natural else [(2, "mass")] if wreck else []):
            value = e.get(name)
            if value is None:
                unknown[CHANNELS[index+6]] += 1
                continue
            value = float(value)
            if not np.isfinite(value) or value < 0:
                raise ValueError("Invalid source entity value")
            fields[index, iz, ix] += value
    before = fields.sum((1, 2))
    for i in range(4):
        fields[i] = gaussian_filter(fields[i], 1.5 if i == 3 else 1., mode="reflect", truncate=3.)
    if not np.allclose(fields.sum((1, 2)), before, rtol=1e-8, atol=1e-7):
        raise ValueError("Entity raster lost quantity")
    return fields.astype(np.float32), dict(unknown), dict(outside), before.tolist()


def transform(physical):
    result = np.asarray(physical, np.float32).copy()
    result[0] = np.arcsinh(result[0]/4.)
    result[1:10] = np.log1p(result[1:10])
    return result


def encode(physical, norm):
    result = transform(physical)
    scales = np.asarray(norm["scales"], np.float32)[:, None, None]
    result[:10] /= scales[:10]
    result[1:] = result[1:]*2-1
    return result.astype(np.float32)


def decode(encoded, norm):
    result = np.asarray(encoded, np.float32).copy()
    scales = np.asarray(norm["scales"], np.float32)[:, None, None]
    result[1:] = (result[1:]+1)*.5
    result[:10] *= scales[:10]
    result[0] = np.sinh(result[0])*4.
    result[1:10] = np.expm1(result[1:10])
    return result


def write_rows(path, rows):
    path.write_text("".join(json.dumps(r, allow_nan=False)+"\n" for r in rows), encoding="utf-8")


def seal(folder, schema, names, **facts):
    write_json(folder/"release_manifest.json", dict(schema=schema, status="ready",
        artifacts={n:file_hash(folder/n) for n in names}, **facts))


def build(root, out, civilians):
    root, out, civilians = map(lambda p: Path(p).resolve(), (root, out, civilians))
    if (out/"release_manifest.json").exists():
        verify_release(out)
        verify_release(civilians)
        print("Existing sealed datasets verified", flush=True)
        return
    if out.exists() or civilians.exists():
        raise ValueError("Choose fresh dataset paths; do not overwrite partial or published releases")
    cohort_path = root/"data/tensors/terrain_curated_20261024_v1"
    props_path = root/"data/tensors/reclaim_props_curated_256_v1"
    texture_path = root/"data/tensors/map_textures_curated_256_v2"
    source_hashes = {str(p.relative_to(root)):verify_release(p) for p in (cohort_path, props_path, texture_path)}
    cohort, props, textures = (joined_rows(p) for p in (cohort_path, props_path, texture_path))
    for other in (props, textures):
        if set(other) != set(cohort) or any((r["split"], r["family"]) !=
            (other[k]["split"], other[k]["family"]) for k,r in cohort.items()):
            raise ValueError("Cohort join or corrected family split differs")
    out.mkdir(parents=True)
    civilians.mkdir(parents=True)
    (out/"maps").mkdir()
    (civilians/"maps").mkdir()
    contract = dict(schema=SCHEMA, source_releases=source_hashes, builder_sha256=file_hash(__file__),
        channels=CHANNELS, ratings=RATINGS, shape=[13, GRID, GRID], spatial_conditions=0,
        generation_inputs="Gaussian noise, diffusion timestep and 12 requested ratings only",
        quantities="Absolute values; reversible asinh/log scaling fitted on training maps; no clipping or per-map normalization",
        excluded_layers="No SDF, passability, front/access/economy layers, supplied anchors or validity-mask model inputs",
        civilian_definition="Source-recorded civilian_unit/building count; runtime activation unresolved",
        natural_definition="Source prop kinds tree/rock/boulder/reclaim, all their known energy and mass",
        water_definition="Saved physical water depth; no shoreline-distance or added water mask",
        source_rgb="Unlit diagnostic albedo RGB, excluding lighting, decals, water rendering and props",
        postprocessing="None; no symmetry, placement repair, count caps or gameplay masks", test_used=False)
    write_json(out/"contract.json", contract)
    packed, civil_rows, excluded = [], [], []
    samples = [[] for _ in CHANNELS]
    started = time.monotonic()
    for index, (key, row) in enumerate(cohort.items(), 1):
        pr, tx = props[key], textures[key]
        source_entities = Path(pr["source_entities"])
        verify(source_entities, pr["source_entities_sha256"])
        with gzip.open(source_entities, "rt", encoding="utf-8") as stream:
            entities = json.load(stream)
        fields, unknown, outside, totals = entity_fields(entities)
        civil_file = civilians/"maps"/(key+".npz")
        np.savez_compressed(civil_file, target=fields[3:4])
        civil_rows.append(dict(key=key, family=row["family"], split=row["split"],
            archive="maps/"+civil_file.name, sha256=file_hash(civil_file), count=totals[3],
            source_entities=str(source_entities), source_entities_sha256=pr["source_entities_sha256"]))
        reasons = []
        if not tx["semantic_weights_valid"] or tx["rgb_valid_fraction"] < 1-1e-6:
            reasons.append("incomplete_or_unsupported_rgb")
        if unknown:
            reasons.extend("unknown_"+name for name in unknown)
        metadata = Path(pr["metadata"])
        verify(metadata, pr["source_metadata_sha256"])
        meta = read_json(metadata)
        if meta["key"] != key or meta["spawns"] != row["spawns"]:
            raise ValueError("Source scenario mismatch: "+key)
        if any(meta["ratings"].get(n) is None or not meta["loss_mask"].get(n) for n in RATINGS):
            reasons.append("incomplete_ratings")
        if reasons:
            excluded.append(dict(key=key, split=row["split"], reasons=reasons, unknown=unknown))
        else:
            native = Path(row["native_source"])
            verify(native, row["native_source_sha256"])
            with np.load(native, allow_pickle=False) as data:
                vertices = data["height_vertices"]
                depth = data["depth_native"]
            if vertices.shape != (513,513) or depth.shape != (512,512):
                raise ValueError("Unexpected native geometry")
            cells = (vertices[:-1,:-1]+vertices[1:,:-1]+vertices[:-1,1:]+vertices[1:,1:])*.25-meta["height_reference"]
            texture = texture_path/tx["archive"]
            verify(texture, tx["sha256"])
            with np.load(texture, allow_pickle=False) as data:
                rgb = data["albedo_rgb"]
                if not bool(data["paint_weights_valid"]) or data["albedo_validity"].min() < 1-1e-6:
                    raise ValueError("RGB metadata disagrees with source array")
            starts = [point_field([p], GRID, sigma=1.5) for p in meta["spawns"]]
            markers = [point_field(meta[n], GRID, sigma=1.5) for n in ("mexes","hydros")]
            physical = np.concatenate([np.stack([pool(cells,GRID),pool(depth,GRID),*starts,*markers]), fields, rgb])
            if physical.shape != (13,GRID,GRID) or not np.isfinite(physical).all() or (physical[1:] < 0).any():
                raise ValueError("Invalid full joint source stack: "+key)
            physical_file = out/"maps"/(key+"_physical.npz")
            np.savez_compressed(physical_file, physical=physical)
            transformed = transform(physical)
            if row["split"] == "train":
                rng = np.random.default_rng(index)
                for i in range(10):
                    values = np.abs(transformed[i]) if i == 0 else transformed[i]
                    positive = values[values>0]
                    if len(positive):
                        samples[i].append(rng.choice(positive, min(512,len(positive)), replace=False))
            packed.append(dict(key=key, family=row["family"], split=row["split"],
                physical_archive="maps/"+physical_file.name, physical_sha256=file_hash(physical_file),
                ratings=[meta["ratings"][n] for n in RATINGS], source_metadata=str(metadata),
                source_metadata_sha256=pr["source_metadata_sha256"],
                native_source=str(native), native_source_sha256=row["native_source_sha256"],
                source_entities=str(source_entities), source_entities_sha256=pr["source_entities_sha256"],
                rgb_source=str(texture), rgb_source_sha256=tx["sha256"],
                totals={"start_a":1.,"start_b":1.,"mex":len(meta["mexes"]),"hydro":len(meta["hydros"]),
                        **dict(zip(CHANNELS[6:10],totals,strict=True))}, outside=outside))
        if index % 50 == 0 or index == len(cohort):
            progress = dict(status="packing_sources", processed=index, total=len(cohort),
                included=len(packed), excluded=len(excluded), elapsed_seconds=time.monotonic()-started)
            write_json(out/"progress.json", progress)
            print(progress, flush=True)
    scales = [max(float(np.quantile(np.concatenate(v), .995)),1e-4) if v else 1. for v in samples[:10]]+[1.]*3
    norm = dict(schema=SCHEMA, channels=CHANNELS, ratings=RATINGS, scales=scales,
        rating_transform="(score - 3) / 2", fitted_on="train_only", clipping=False,
        sampling_min=[float("inf")]*13, sampling_max=[-float("inf")]*13)
    max_error = 0.
    for row in packed:
        with np.load(out/row["physical_archive"],allow_pickle=False) as data:
            physical=data["physical"]
        target=encode(physical,norm)
        restored=decode(target,norm)
        max_error=max(max_error,float(np.max(np.abs(restored-physical)/(1+np.abs(physical)))))
        if not np.allclose(restored,physical,rtol=3e-5,atol=2e-4):
            raise ValueError("Joint normalization roundtrip failed")
        if row["split"]=="train":
            norm["sampling_min"]=np.minimum(norm["sampling_min"],target.min((1,2))).tolist()
            norm["sampling_max"]=np.maximum(norm["sampling_max"],target.max((1,2))).tolist()
        file=out/"maps"/(row["key"]+".npz")
        profile=(np.asarray(row["ratings"],np.float32)-3)/2
        np.savez_compressed(file,target=target,ratings=profile)
        row.update(archive="maps/"+file.name,sha256=file_hash(file))
    for folder, records in ((out,packed),(civilians,civil_rows)):
        for split in ("train","validation"):
            write_rows(folder/(split+".jsonl"),[r for r in records if r["split"]==split])
    write_json(out/"normalization.json",norm)
    write_json(out/"excluded.json",excluded)
    audit=dict(status="passed",maps=len(packed),splits=dict(Counter(r["split"] for r in packed)),
        families=len({r["family"] for r in packed}),shape=[13,GRID,GRID],ratings=RATINGS,
        positive_maps={name:sum(r["totals"].get(name,0)>0 for r in packed) for name in CHANNELS[2:10]},
        exclusion_reasons=dict(Counter(reason for r in excluded for reason in r["reasons"])),
        max_relative_roundtrip_error=max_error,source_hashes_verified=True,test_used=False,
        elapsed_seconds=time.monotonic()-started)
    if len({r["family"] for r in packed})!=len(packed) or not all(audit["splits"].get(s,0)>0 for s in ("train","validation")):
        raise ValueError("Family independence or split coverage failed")
    write_json(out/"audit.json",audit)
    write_json(out/"progress.json",dict(status="ready",**{k:v for k,v in audit.items() if k!="status"}))
    seal(out,SCHEMA,["contract.json","normalization.json","train.jsonl","validation.jsonl","excluded.json","audit.json"],maps=len(packed))
    write_json(civilians/"contract.json",dict(schema="source_civilian_density_v1",shape=[1,GRID,GRID],
        units="Source-recorded entities per cell; Gaussian sigma 1.5 model cells; sum equals count",
        source_releases=source_hashes,definition=contract["civilian_definition"],builder_sha256=file_hash(__file__)))
    write_json(civilians/"audit.json",dict(status="passed",maps=len(civil_rows),
        splits=dict(Counter(r["split"] for r in civil_rows)),
        positive_maps=dict(Counter(r["split"] for r in civil_rows if r["count"]>0)),
        count=sum(r["count"] for r in civil_rows),no_per_map_normalization=True))
    seal(civilians,"source_civilian_density_v1",["contract.json","audit.json","train.jsonl","validation.jsonl"],maps=len(civil_rows))
    print(json.dumps(audit,indent=2),flush=True)


if __name__=="__main__":
    parser=argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--root",type=Path,default=Path.cwd())
    parser.add_argument("--out",type=Path,required=True)
    parser.add_argument("--civilians",type=Path,required=True)
    args=parser.parse_args()
    with lease(args.root/".local-tools/joint-source-build.lock"):
        build(args.root,args.out,args.civilians)
