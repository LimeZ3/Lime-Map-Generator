def read(p):
    p=Path(p)
    return json.loads(gzip.open(p,'rt',encoding='utf-8').read() if p.suffix=='.gz' else p.read_text(encoding='utf-8-sig'))
def rel(p):
    try:return Path(p).relative_to(ROOT).as_posix()
    except ValueError:return Path(p).name
def sha(p):return hashlib.sha256(Path(p).read_bytes()).hexdigest()
def js(p,prefix,data):p.write_text(prefix+json.dumps(finite(data),ensure_ascii=False,allow_nan=False,separators=(',',':')).replace('<','\\u003c')+';\n',encoding='utf-8')
def finite(v):
    if isinstance(v,str) and len(v)>2 and v[1]==':' and v[2] in ('/','\\'):return rel(v)
    if isinstance(v,dict):return {k:finite(x) for k,x in v.items()}
    if isinstance(v,(list,tuple)):return [finite(x) for x in v]
    if isinstance(v,(float,np.floating)):return float(v) if math.isfinite(v) else None
    if isinstance(v,np.integer):return int(v)
    return v

def point(d,p,color,r=2,label=None):
    x,z=p.get('x',-1)/2,p.get('z',-1)/2
    if not (0<=x<256 and 0<=z<256):return
    d.ellipse((x-r,z-r,x+r,z+r),fill=(*color,255))
    if label:d.text((min(220,x+5),min(242,z+2)),label,fill=(*color,255),stroke_width=1,stroke_fill='black')
def raster(array,mode,color):

def find_paths(value):
    # Only explicit named point polylines. Never reinterpret scalar arrays as paths.
    found=[]
    if isinstance(value,dict):
        for k,v in value.items():
            if k in ('path','polyline','route_points') and isinstance(v,list) and len(v)>1:
                if all(isinstance(p,dict) and 'x' in p and 'z' in p for p in v):found.append([(p['x'],p['z']) for p in v])
            elif isinstance(v,(dict,list)):found.extend(find_paths(v))
    elif isinstance(value,list):
        for item in value:
            if isinstance(item,(dict,list)):found.extend(find_paths(item))
    return found
