"""Five-scale terrain U-Net; separate from the historical three-scale baseline."""

from __future__ import annotations

import math

import torch
from torch import nn
from torch.utils.checkpoint import checkpoint

from supmapai.ml.train_terrain_ddpm import SinusoidalTimeEmbedding


class TerrainBlock(nn.Module):
    def __init__(self, inputs, outputs, embedding, dropout):
        super().__init__()
        self.norm1 = nn.GroupNorm(math.gcd(inputs, 8), inputs)
        self.conv1 = nn.Conv2d(inputs, outputs, 3, padding=1)
        self.norm2 = nn.GroupNorm(math.gcd(outputs, 8), outputs)
        self.affine = nn.Linear(embedding, outputs * 2)
        self.dropout = nn.Dropout2d(dropout)
        self.conv2 = nn.Conv2d(outputs, outputs, 3, padding=1)
        self.skip = nn.Conv2d(inputs, outputs, 1) if inputs != outputs else nn.Identity()

    def forward(self, x, embedding):
        hidden = self.conv1(nn.functional.silu(self.norm1(x)))
        scale, shift = self.affine(nn.functional.silu(embedding)).chunk(2, dim=1)
        hidden = self.norm2(hidden) * (1 + scale[:, :, None, None]) + shift[:, :, None, None]
        return self.skip(x) + self.conv2(self.dropout(nn.functional.silu(hidden)))


class SpatialAttention(nn.Module):
    def __init__(self, channels):
        super().__init__()
        self.norm = nn.LayerNorm(channels)
        self.attention = nn.MultiheadAttention(channels, 4, batch_first=True)

    def forward(self, x):
        tokens = x.flatten(2).transpose(1, 2)
        normalized = self.norm(tokens)
        result = self.attention(normalized, normalized, normalized, need_weights=False)[0]
        return (tokens + result).transpose(1, 2).reshape(x.shape)


class MultiscaleTerrainUNet(nn.Module):
    def __init__(
        self, base=64, profile_dim=21, embedding=256, dropout=0.1, activation_checkpointing=True
    ):
        super().__init__()
        self.activation_checkpointing = activation_checkpointing
        self.time = nn.Sequential(
            SinusoidalTimeEmbedding(embedding),
            nn.Linear(embedding, embedding),
            nn.SiLU(),
            nn.Linear(embedding, embedding),
        )
        self.profile = nn.Sequential(
            nn.Linear(profile_dim, embedding), nn.SiLU(), nn.Linear(embedding, embedding)
        )
        channels = [base * m for m in (1, 2, 3, 4, 4)]
        self.input = nn.Conv2d(5, channels[0], 3, padding=1)
        self.encoders = nn.ModuleList(
            [
                nn.ModuleList([TerrainBlock(c, c, embedding, dropout) for _ in range(2)])
                for c in channels
            ]
        )
        self.down = nn.ModuleList(
            [
                nn.Conv2d(a, b, 3, stride=2, padding=1)
                for a, b in zip(channels[:-1], channels[1:], strict=True)
            ]
        )
        self.attention = nn.ModuleDict({str(i): SpatialAttention(channels[i]) for i in (3, 4)})
        self.middle = TerrainBlock(channels[-1], channels[-1], embedding, dropout)
        self.ups = nn.ModuleList()
        self.decoders = nn.ModuleList()
        for i in range(len(channels) - 2, -1, -1):
            self.ups.append(nn.Conv2d(channels[i + 1], channels[i], 3, padding=1))
            self.decoders.append(
                nn.ModuleList(
                    [
                        TerrainBlock(channels[i] * 2, channels[i], embedding, dropout),
                        TerrainBlock(channels[i], channels[i], embedding, dropout),
                    ]
                )
            )
        self.output = nn.Sequential(
            nn.GroupNorm(math.gcd(base, 8), base), nn.SiLU(), nn.Conv2d(base, 3, 3, padding=1)
        )

    def run_block(self, block, *args):
        if self.activation_checkpointing and self.training and torch.is_grad_enabled():
            return checkpoint(block, *args, use_reentrant=False)
        return block(*args)

    def forward(self, noisy, timesteps, spatial, profile):
        embedding = self.time(timesteps) + self.profile(profile)
        x = self.input(torch.cat([noisy, spatial], dim=1))
        skips = []
        for i, blocks in enumerate(self.encoders):
            for block in blocks:
                x = self.run_block(block, x, embedding)
            if str(i) in self.attention:
                x = self.run_block(self.attention[str(i)], x)
            skips.append(x)
            if i < len(self.down):
                x = self.down[i](x)
        x = self.run_block(self.middle, x, embedding)
        for up, blocks, skip in zip(self.ups, self.decoders, reversed(skips[:-1]), strict=True):
            x = up(
                nn.functional.interpolate(
                    x, size=skip.shape[-2:], mode="bilinear", align_corners=False
                )
            )
            x = torch.cat([x, skip], dim=1)
            for block in blocks:
                x = self.run_block(block, x, embedding)
        return self.output(x)
