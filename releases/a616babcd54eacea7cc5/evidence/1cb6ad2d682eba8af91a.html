def replace_retry(source,target):
    for attempt in range(24):
        try:source.replace(target);return
        except PermissionError:
            if attempt==23:raise
            time.sleep(.25)

def dump(path,value):
    path.parent.mkdir(parents=True,exist_ok=True)
    tmp=path.with_name(path.name+'.'+str(os.getpid())+'.tmp')
    tmp.write_text(json.dumps(value,indent=2,allow_nan=False),encoding='utf-8');replace_retry(tmp,path)

def select_pair(nav):
    a=next((s for s in nav['spawns'] if s['name']=='ARMY_1'),None)
    if a is None:raise ValueError('No slot 1')
    w,h=nav['width'],nav['height'];options=[]
    targets=[('rotate180_approximate',(w-a['x'],h-a['z'])),('mirror_x',(w-a['x'],a['z'])),('mirror_z',(a['x'],h-a['z']))]
    if w==h:targets += [('diagonal',(a['z'],a['x'])),('anti_diagonal',(w-a['z'],h-a['x']))]
    for kind,target in targets:
        for b in nav['spawns']:
            if b['name']==a['name']:continue
            error=math.hypot(b['x']-target[0],b['z']-target[1]);distance=math.hypot(a['x']-b['x'],a['z']-b['z'])
            # A precise reflection must outrank a merely approximate rotation.
            # Treat sub-cell differences as ties; prefer rotation only on a tie.
            if error<=max(4,.05*min(w,h)):options.append((max(0,error-1),0 if kind=='rotate180_approximate' else 1,error,-distance,b['name'],b,kind))
    if options:
        _,_,error,_,_,b,kind=min(options)
        if kind=='rotate180_approximate' and error<=max(4,.02*min(w,h)):kind='rotate180'
        return a,b,dict(kind=kind,position_error=error,selection_policy='closest geometric counterpart; sub-cell ties prefer rotation',
            candidates=[dict(slot=o[-2]['name'],kind=o[-1],position_error=o[2]) for o in sorted(options)])
    others=[s for s in nav['spawns'] if s['name']!=a['name']]
    if len(others)==1:return a,others[0],dict(kind='only_other_slot_asymmetric',position_error=None,fallback='two-slot map; no geometric mirror assumed')
    raise ValueError('No unambiguous mirrored opponent for slot 1')

def front_metrics(d):
    routes=d['routes']
    if not routes:return dict(category='disconnected',route_count=0,best_bottleneck=None,median_bottleneck=None,mean_route_width=None,all_route_shared_length=0,search_limits=d.get('limits',[]))
    widths=[r['width']['minimum_sustained_8'] for r in routes if r.get('width')]
    best=max(widths);shared=sum(s['length'] for s in d.get('merged_sections',[]) if len(s['route_ids'])==len(routes)) if len(routes)>1 else 0
    return dict(category='closed' if best<24 else 'open' if best>=64 else 'mixed',route_count=len(routes),
        route_family_count=len({tuple(r['signature']) for r in routes}),best_bottleneck=best,median_bottleneck=float(np.median(widths)),mean_route_width=float(np.mean([r['width']['mean'] for r in routes])),
        shortest_length=min(r['length'] for r in routes),longest_length=max(r['length'] for r in routes),
        all_route_shared_length=shared,search_limits=d.get('limits',[]),
        widths=[dict(route=r['id'],minimum=r['width']['minimum'],sustained_8=r['width']['minimum_sustained_8'],mean=r['width']['mean']) for r in routes])

def resolve_configuration(key,folder,script,nav,slots):
    resolution=adaptive_resources.resolve(folder,script,[dict(**s,slot=int(s['name'].split('_')[-1]),slot_valid=True) for s in nav['spawns']],
        [m for m in nav['markers'] if m['type']=='Mass'],[m for m in nav['markers'] if m['type']=='Hydrocarbon'],slots)
    if resolution['status']=='resolved_supported_static_rules':return resolution
    _,old=effective_resources(folder,nav,script)
    if old['status']=='static_save_markers':return dict(status='static_save_markers',active_slots=slots,selected_options={})
    path=ROOT/'data/manifests/strategy_static_resource_reviews.json'
    review=json.loads(path.read_text(encoding='utf-8')).get(key) if path.exists() else None
    if review:
        for name,expected in review['files'].items():
            if hashlib.sha256((folder/name).read_bytes()).hexdigest()!=expected:raise ValueError('Reviewed static helper changed: '+name)
        return dict(status='static_save_markers',active_slots=slots,selected_options={},review=review)
    raise ValueError('Unresolved default adaptive resources: '+'; '.join(resolution['warnings']))

def error_class(message):
    if 'mirrored opponent' in message:return 'no_mirrored_1v1_pair'
    if 'Unresolved default adaptive resources' in message:return 'unsupported_resource_script'
    if 'memory_budget' in message:return 'memory_limit'
    if 'timeout' in message:return 'timeout'
    if 'worker_exit' in message:return 'worker_exit'
    return 'extraction_or_analysis_error'

def relative_rating(value,values):
    if value<=0:return 1.
    # Midrank percentile, tied values get equal scores; zero always means minimum.
    rank=(sum(v<value for v in values)+.5*sum(v==value for v in values))/len(values)
    return round(1+4*rank,2)

def process_alive(pid):
    import ctypes
    kernel=ctypes.windll.kernel32;kernel.OpenProcess.restype=ctypes.c_void_p
    kernel.GetExitCodeProcess.argtypes=[ctypes.c_void_p,ctypes.POINTER(ctypes.c_ulong)]
    kernel.CloseHandle.argtypes=[ctypes.c_void_p]
    handle=kernel.OpenProcess(0x1000,False,int(pid))
    if not handle:return None if kernel.GetLastError()==5 else False
    code=ctypes.c_ulong()
    try:return bool(kernel.GetExitCodeProcess(handle,ctypes.byref(code))) and code.value==259
    finally:kernel.CloseHandle(handle)

def probe():
    path=OUT/'status.json';state=json.loads(path.read_text(encoding='utf-8')) if path.exists() else {}
    state['driver_alive']=process_alive(state['pid']) if state.get('pid') else False
    state['heartbeat_age_seconds']=time.time()-state.get('heartbeat',0)
    state['driver_recent_heartbeat']=state.get('state')=='running' and state['heartbeat_age_seconds']<90
    for r in state.get('running',[]):
        r['worker_alive']=process_alive(r['pid']) if r.get('pid') else None
        stage_path=OUT/r['key']/'stage.json';current=json.loads(stage_path.read_text(encoding='utf-8')) if stage_path.exists() else {}
        terminal=(OUT/r['key']/'summary.json').exists() or current.get('stage') in {'failed','complete'}
        if terminal:r['worker_alive']=False
        r['may_be_in_progress']=not terminal and (r['worker_alive'] is True or r['worker_alive'] is None and time.time()-r.get('updated',0)<2400)
    if (OUT/'report.json').exists():
        d=json.loads((OUT/'report.json').read_text(encoding='utf-8'))
        state['coverage']={k:d[k] for k in ['total','complete','failed','pending']}
    return state

def available_gb():
    import ctypes
    class Mem(ctypes.Structure):_fields_=[('length',ctypes.c_ulong),('load',ctypes.c_ulong),*[(n,ctypes.c_ulonglong) for n in ['total','avail','page','apage','v','av','ex']]]
    m=Mem();m.length=ctypes.sizeof(m);ctypes.windll.kernel32.GlobalMemoryStatusEx(ctypes.byref(m));return m.avail/2**30

def run(workers=2,limit=None,retry=False,only_keys=None):
    OUT.mkdir(parents=True,exist_ok=True)
    previous=probe()
    if (previous.get('driver_alive') or previous.get('driver_recent_heartbeat')) and previous.get('pid')!=os.getpid():
        print('Existing controller is running:',previous['pid'],flush=True);return
    if any(r.get('may_be_in_progress') for r in previous.get('running',[])):
        print('Previous workers are still running; wait for them before resuming.',flush=True);return
    items=json.loads(COHORT.read_text(encoding='utf-8'))['maps'];sig=fingerprint()
    if only_keys:items=[m for m in items if m['key'] in set(only_keys)]
    if limit:items=items[:limit]
    pending=[]
    for m in items:
        out=OUT/m['key'];done=out/'summary.json';failed=out/'failure.json'
        if done.exists():continue
        if failed.exists() and not retry:continue
        pending.append(m)
    active=[];started=time.time();last_report=0
    while pending or active:
        # Keep large maps exclusive. Each map is isolated in a fresh process.
        if pending and len(active)<workers:
            m=pending[0];cells=m['width']*m['height'];need=max(1.5,cells/1048576*.75)
            memory=available_gb()
            if not active and cells>=4*1048576 and memory<need:
                # Explicit resource-bound failure instead of risking machine-wide OOM.
                dump(OUT/m['key']/'failure.json',dict(**m,status='failed',error=f'memory_budget: estimated {need:.1f} GiB free required, {memory:.1f} available',code_sha256=sig));pending.pop(0)
