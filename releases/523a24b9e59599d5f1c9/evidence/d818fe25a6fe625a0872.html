"""Freeze human exclusions and one representative per terrain family for new runs."""

from __future__ import annotations

import argparse
import copy
import json
import re
from collections import Counter, defaultdict
from pathlib import Path

import numpy as np

from supmapai.strategy.pipeline_contract import digest, dump, file_hash, lease
from supmapai.terrain_controls.half_targets import encode_targets, fit_scaling
from supmapai.tools.dataset_review import load_cohort


def name_key(folder):
    value = re.sub(r"\.v\d+(?:\.\d+)*$", "", folder.casefold())
    value = re.sub(r"\b\d+v\d+\b|\b\d+x\d+\b", " ", value.replace("_", " "))
    value = re.sub(r"\bv\d+\b", " ", value)
    value = re.sub(r"\b(adaptive|flooded|dry|fixed|edition|faf|version|teamplay|ai)\b", " ", value)
    return re.sub(r"[^a-z0-9]", "", value)


def version_rank(entry):
    folder = entry["folder"].casefold()
    match = re.search(r"\.v(\d+(?:\.\d+)*)$", folder)
    version = tuple(map(int, match[1].split("."))) if match else (0,)
    # Author versions take priority over vault upload IDs, which are not version numbers.
    explicit = [int(x) for x in re.findall(r"(?:^|[ _-])v(\d+)(?:$|[ ._-])", folder)]
    return (
        bool(re.search(r"(?:^|[ _-])faf(?:$|[ ._-])", folder)),
        version,
        max(explicit, default=0),
        "adaptive" in folder,
        int(entry["key"].split("_")[-1]) if entry["key"].split("_")[-1].isdigit() else 0,
        entry["key"],
    )


class Groups:
    def __init__(self, keys):
        self.parent = {k: k for k in keys}

    def root(self, key):
        if self.parent[key] != key:
            self.parent[key] = self.root(self.parent[key])
        return self.parent[key]

    def union(self, a, b):
        a, b = self.root(a), self.root(b)
        self.parent[max(a, b)] = min(a, b)


def select(rows, entries, decisions, geometry_pairs=()):
    groups = Groups(r["key"] for r in rows)
    seen = {}
    evidence = []
    for row in rows:
        key, entry = row["key"], entries[row["key"]]
        tokens = [("source_family", row["family"])]
        normalized = name_key(entry["folder"])
        if len(normalized) >= 4:
            tokens.append(("normalized_name", normalized))
        map_id = key.split("_")[0]
        if map_id.isdigit() and int(map_id) > 0:
            tokens.append(("vault_map", map_id))
        # Explicit user-requested alias: do not generalize this to gap/pass/crater.
        if "canis" in entry["folder"].casefold():
            tokens.append(("confirmed_alias", "canis"))
        for kind, value in tokens:
            token = kind, value
            if token in seen:
                other = seen[token]
                groups.union(key, other)
                evidence.append(dict(a=other, b=key, kind=kind, value=value))
            else:
                seen[token] = key
    for pair in geometry_pairs:
        groups.union(pair["a"], pair["b"])
        evidence.append(dict(pair, kind="near_identical_height"))
    members = defaultdict(list)
    for row in rows:
        members[groups.root(row["key"])].append(row)
    selected, families, omitted = [], [], []
    for group in members.values():
        candidates = [r for r in group if decisions.get(r["key"], {}).get("status") != "exclude"]
        if not candidates:
            continue
        winner = max(candidates, key=lambda r: version_rank(entries[r["key"]]))
        # If a newly recognized family crosses splits, all of it is held out.
        split = "validation" if any(r["split"] == "validation" for r in group) else "train"
        family = "curated_" + digest(sorted(r["key"] for r in group))[:16]
        selected.append(
            dict(
                winner,
                family=family,
                split=split,
                original_family=winner["family"],
                original_split=winner["split"],
            )
        )
        families.append(
            dict(
                family=family,
                selected=winner["key"],
                split=split,
                selected_name=entries[winner["key"]]["folder"],
                members=[
                    dict(
                        key=r["key"],
                        folder=entries[r["key"]]["folder"],
                        original_split=r["split"],
                        manually_excluded=decisions.get(r["key"], {}).get("status") == "exclude",
                    )
                    for r in group
                ],
            )
        )
        for row in candidates:
            if row["key"] != winner["key"]:
                omitted.append(
                    dict(key=row["key"], selected=winner["key"], reason="superseded_family_member")
                )
    return selected, dict(families=families, omitted=omitted, merge_evidence=evidence)


def geometry_duplicates(source, rows):
    """Conservative high-correlation check, invariant to eight grid orientations.

    This is for almost identical geometry, not broad thematic resemblance. Flat
    or near-constant maps are never matched just because both have little relief.
    """
    features, eligible = [], []
    for row in rows:
        path = Path(source) / row["archive"]
        if file_hash(path) != row["sha256"]:
            raise ValueError("Source target checksum changed: " + row["key"])
        with np.load(path, allow_pickle=False) as archive:
            height = archive["target"][0].copy()
        # Asinh height is monotonic; affine offsets/scales are removed on this
        # representation only. Require sub-percent agreement, not loose similarity.
        low = height.reshape(32, 8, 32, 8).mean(axis=(1, 3))
        low = low - low.mean()
        norm = float(np.linalg.norm(low))
        eligible.append(norm > 0.05 and len(np.unique(np.round(low, 4))) > 32)
        features.append(low / max(norm, 1e-8))
    base = np.stack(features).astype(np.float32)
    best = np.full((len(rows), len(rows)), -1.0, dtype=np.float32)
    for rotation in range(4):
        rotated = np.rot90(base, rotation, axes=(-2, -1))
        for flip in (False, True):
            variant = rotated[..., ::-1] if flip else rotated
            score = base.reshape(len(rows), -1) @ variant.reshape(len(rows), -1).T
            np.maximum(best, score, out=best)
    pairs = []
    for i, j in zip(*np.where(np.triu(best >= 0.9998, 1)), strict=True):
        if eligible[i] and eligible[j]:
            pairs.append(dict(a=rows[i]["key"], b=rows[j]["key"], correlation=float(best[i, j])))
    return pairs


def build(source, review_dir, out):
    source, cohort, rows = load_cohort(source)
    review_dir, out = Path(review_dir).resolve(), Path(out).resolve()
    decisions_path = review_dir / "decisions.json"
    state = json.loads(decisions_path.read_text(encoding="utf-8"))
    catalog = json.loads((review_dir / "catalog.json").read_text(encoding="utf-8"))
    if state["cohort"] != cohort or catalog["cohort"] != cohort:
        raise ValueError("Review/source cohort mismatch")
    entries = {m["key"]: m for m in catalog["maps"]}
    identity = dict(
        source=str(source),
        source_release_sha256=cohort,
        review_sha256=file_hash(decisions_path),
        code_sha256=file_hash(__file__),
    )
    out.mkdir(parents=True, exist_ok=True)
    with lease(out / "build.lock"):
        if (out / "release_manifest.json").exists():
            if json.loads((out / "curation_contract.json").read_text()) != identity:
                raise ValueError("Curation inputs changed; choose a new output directory")
            load_cohort(out)
            return json.loads((out / "audit.json").read_text())
        print("Checking near-identical terrain across orientations...", flush=True)
        pairs = geometry_duplicates(source, rows)
        selected, report = select(rows, entries, state["decisions"], pairs)
        if not all(any(r["split"] == s for r in selected) for s in ("train", "validation")):
            raise ValueError("Curation would leave an empty split")
        controls = fit_scaling([dict(split=r["split"], **r["targets"]) for r in selected])
        norm = json.loads((source / "normalization.json").read_text())
        norm = copy.deepcopy(norm)
        norm.update(
            controls=controls,
            training_schema="terrain_curated_overnight_v1",
            control_scaling="refitted_on_retained_train_only",
            physical_encoding="frozen_source_physical_units_for_comparable_targets",
        )
        # Target encodings are fixed physical coordinates. Refit condition scaling,
        # not terrain values: this preserves measured labels and avoids resampling.
        for row in selected:
            row["archive"] = str((source / row["archive"]).resolve())
            row["profile"] = encode_targets(row["targets"], controls, row["water_enabled"]).tolist()
        for split in ("train", "validation"):
            path = out / (split + ".jsonl")
            path.write_text(
                "".join(
                    json.dumps(r, allow_nan=False) + "\n" for r in selected if r["split"] == split
                ),
                encoding="utf-8",
            )
        audit = dict(
            status="passed",
            maps=len(selected),
            splits=dict(Counter(r["split"] for r in selected)),
            manually_excluded=sum(v["status"] == "exclude" for v in state["decisions"].values()),
            superseded=len(report["omitted"]),
            near_identical_pairs=len(pairs),
            cross_split_families=[
                g
                for g in report["families"]
                if len({m["original_split"] for m in g["members"]}) > 1
            ],
            checks=[
                "Human exclusions preserved; unreviewed retained",
                "One map per merged family",
                "FAF-labelled version preferred, otherwise highest author version",
                "Cross-split merged families held out",
                "Control scaling fit on retained train only",
                "Target checksums verified; no original maps or tensors modified",
            ],
            test_used=False,
        )
        dump(out / "normalization.json", norm)
        dump(out / "curation_contract.json", identity)
        dump(out / "review_snapshot.json", state)
        dump(out / "family_selection.json", report)
        dump(out / "audit.json", audit)
        artifacts = [
            "normalization.json",
            "train.jsonl",
            "validation.jsonl",
            "curation_contract.json",
            "review_snapshot.json",
            "family_selection.json",
            "audit.json",
        ]
        dump(
            out / "release_manifest.json",
            dict(
                schema="terrain_full256_controls_v3",
                status="ready",
                artifacts={p: file_hash(out / p) for p in artifacts},
            ),
        )
        load_cohort(out)
        return audit


def main():
    p = argparse.ArgumentParser(description=__doc__)
    p.add_argument("--source", type=Path, required=True)
    p.add_argument("--review-dir", type=Path, required=True)
    p.add_argument("--out", type=Path, required=True)
    args = p.parse_args()
    print(json.dumps(build(args.source, args.review_dir, args.out), indent=2))


if __name__ == "__main__":
    main()
