class RatingEmbedding(nn.Module):
    def __init__(self, width, drop):
        super().__init__()
        self.net=nn.Sequential(nn.Linear(12,width),nn.SiLU(),nn.Linear(width,width))
        self.null=nn.Parameter(torch.zeros(1,width))
        self.drop=drop

    def forward(self, ratings):
        if ratings is None:
            return self.null
        result=self.net(ratings)
        if self.training and self.drop:
            keep=torch.rand((len(ratings),1),device=ratings.device)>=self.drop
            result=torch.where(keep,result,self.null)
        return result


class JointUNet(MultiscaleTerrainUNet):
    def __init__(self,base=128,dropout=.1,checkpointing=False):
        super().__init__(base=base,profile_dim=12,embedding=384,dropout=dropout,
                         activation_checkpointing=checkpointing)
        self.input=nn.Conv2d(13,base,3,padding=1)
        self.output[-1]=nn.Conv2d(base,13,3,padding=1)
        self.profile=RatingEmbedding(384,.1 if dropout else 0.)

    def forward(self,noisy,timesteps,ratings):
        empty=noisy.new_empty((len(noisy),0,*noisy.shape[-2:]))
        return super().forward(noisy,timesteps,empty,ratings)


class JointDataset:
    def __init__(self,root,split,limit=None):
        self.root=Path(root)
        verify_release(self.root)
        self.norm=read_json(self.root/"normalization.json")
        self.rows=read_rows(self.root/(split+".jsonl"))
        if limit:
            # Exercise rare positive layers in the implementation test, not the full run.
            rng=np.random.default_rng(20261025)
            selected=[]
            for name in ("civilians","wreck_mass"):
                candidates=[i for i,r in enumerate(self.rows) if r["totals"][name]>0]
                selected.extend(rng.permutation(candidates)[:max(2,limit//4)].tolist())
            selected=list(dict.fromkeys(selected))
            selected.extend(i for i in rng.permutation(len(self.rows)).tolist() if i not in selected)
            self.rows=[self.rows[i] for i in selected[:limit]]
        targets,profiles=[],[]
        for i,row in enumerate(self.rows):
            path=self.root/row["archive"]
            if file_hash(path)!=row["sha256"]:
                raise ValueError("Joint training archive changed: "+row["key"])
            with np.load(path,allow_pickle=False) as data:
                x=data["target"]
                p=data["ratings"]
            if x.shape!=(13,256,256) or p.shape!=(12,) or not np.isfinite(x).all():
                raise ValueError("Invalid joint training packet")
            targets.append(x.astype(np.float16))
            profiles.append(p)
            if (i+1)%100==0:
                print(f'cached {split} {i+1}/{len(self.rows)}',flush=True)
        self.targets=np.stack(targets)
        self.profiles=np.stack(profiles).astype(np.float32)
        if not len(self.rows):
            raise ValueError("Empty joint dataset split")

    def batch(self,indices,device,rng=None):
        target=self.targets[np.asarray(indices)].copy()
        profiles=self.profiles[np.asarray(indices)].copy()
        if rng is not None:
            rotation=int(rng.integers(4))
            target=np.rot90(target,rotation,axes=(-2,-1))
            if rng.random()<.5:
                target=target[...,::-1]
            target=target.copy()
            if rng.random()<.5:
                target[:,[2,3]]=target[:,[3,2]]
        return (torch.from_numpy(np.ascontiguousarray(target)).to(device=device,dtype=torch.float32),
                torch.from_numpy(profiles).to(device))


def loss_terms(model,clean,ratings,cumulative,generator,timestep=None):
    t=(torch.randint(len(cumulative),(len(clean),),device=clean.device,generator=generator)
       if timestep is None else torch.full((len(clean),),timestep,device=clean.device,dtype=torch.long))
    alpha=cumulative[t,None,None,None]
    noise=torch.randn(clean.shape,device=clean.device,generator=generator)
    noisy=alpha.sqrt()*clean+(1-alpha).sqrt()*noise
    prediction=model(noisy,t,ratings).float()
    velocity=alpha.sqrt()*noise-(1-alpha).sqrt()*clean
    # Target-dependent emphasis, not supplied masks or analytical spatial layers.
    weights=torch.ones_like(clean)
    weights[:,2:10]=1+12*((clean[:,2:10]+1)*.5).clamp(0,1)
    error=(prediction-velocity).square()
    per_channel=(error*weights).mean((2,3))/weights.mean((2,3))
    estimate=alpha.sqrt()*noisy-(1-alpha).sqrt()*prediction
    reconstruction=(estimate-clean).square().mean((2,3))
    return per_channel,reconstruction


@torch.no_grad()
def validate(model,data,cumulative,batch_size,seed=61025):
    model.eval()
    gen=torch.Generator(device="cuda").manual_seed(seed)
    values,recon=[],[]
    for start in range(0,len(data.rows),batch_size):
        clean,ratings=data.batch(range(start,min(start+batch_size,len(data.rows))),"cuda")
        with torch.autocast("cuda",dtype=torch.bfloat16):
            loss,mse=loss_terms(model,clean,ratings,cumulative,gen,timestep=500)
        values.extend(loss.cpu().tolist())
        recon.extend(mse.cpu().tolist())
    by_channel=np.mean(values,axis=0)
    return dict(validation_loss=float(by_channel.mean()),
        channel_velocity_loss=dict(zip(CHANNELS,by_channel.tolist(),strict=True)),
        channel_reconstruction_mse=dict(zip(CHANNELS,np.mean(recon,axis=0).tolist(),strict=True)),
        meaning="Fixed timestep 500 denoising of held-out noised sources; not fresh generation")


@torch.no_grad()
def sample(model,ratings,norm,seed=303,steps=64,guidance=1.5,size=256):
    model.eval()
    device=next(model.parameters()).device
    rng=np.random.default_rng(seed)
    x=torch.from_numpy(rng.standard_normal((len(ratings),13,size,size),dtype=np.float32)).to(device)
    cumulative=schedule(1000,device)
    times=np.linspace(999,0,steps,dtype=int)
    lower=np.asarray(norm["sampling_min"],np.float32)
    upper=np.asarray(norm["sampling_max"],np.float32)
    span=np.maximum(upper-lower,.1)
    lower[0]-=.2*span[0]
    upper[:10]+=.2*span[:10]
    low=torch.as_tensor(lower,device=device)[None,:,None,None]
    high=torch.as_tensor(upper,device=device)[None,:,None,None]
    for i,step in enumerate(times):
        t=torch.full((len(x),),int(step),device=device,dtype=torch.long)
        with torch.autocast(device.type,dtype=torch.bfloat16,enabled=device.type=="cuda"):
            velocity=model(x,t,ratings).float()
            if guidance!=1:
                baseline=model(x,t,None).float()
                velocity=baseline+guidance*(velocity-baseline)
        alpha=cumulative[step]
        raw=alpha.sqrt()*x-(1-alpha).sqrt()*velocity
        # Numerical sampling bounds only; retain the final unbounded estimate separately.
        clean=raw.maximum(low).minimum(high)
        epsilon=alpha.sqrt()*velocity+(1-alpha).sqrt()*x
        previous=cumulative[times[i+1]] if i+1<len(times) else torch.ones((),device=device)
        x=previous.sqrt()*clean+(1-previous).sqrt()*epsilon
    return raw.cpu().numpy(),x.cpu().numpy()


def layer_strip(encoded,norm,title):
    size=144
    canvas=Image.new("RGB",(11*size,188),"#f0f2f0")
    draw=ImageDraw.Draw(canvas)
    draw.text((5,4),title,fill="#15251b")
    physical=decode(encoded,norm)
    names=CHANNELS[:10]+["surface RGB"]
    colors=[(210,210,210),(40,135,245),(65,145,255),(245,135,55),(240,190,45),
            (40,220,225),(70,205,90),(205,145,65),(175,105,220),(240,100,180)]
    for i,name in enumerate(names):
        if i==10:
            rgb=np.moveaxis(physical[10:13],0,-1).clip(0,1)
        else:
            strength=((encoded[i]+1)/2).clip(0,1)
            if i==0:
                lo,hi=norm["sampling_min"][0],norm["sampling_max"][0]
                strength=((encoded[0]-lo)/max(hi-lo,1e-4)).clip(0,1)
            rgb=strength[...,None]*np.asarray(colors[i])/255.
        tile=Image.fromarray(np.uint8(rgb*255)).resize((size-4,size-4),Image.Resampling.LANCZOS)
        canvas.paste(tile,(i*size+2,42))
        draw.text((i*size+4,25),name,fill="#15251b")
    return canvas


@torch.no_grad()
def gallery(model,data,out,step,norm,batch_size=2,full_sweep=False):
    folder=Path(out)/"samples"/f"step_{step:07d}"
    folder.mkdir(parents=True,exist_ok=True)
    typical=np.median(data.profiles,axis=0).astype(np.float32)
    requests=[("seed303_typical",303,typical.copy()),("seed404_typical",404,typical.copy())]
    sweep=range(12) if full_sweep else [7]
    for i in sweep:
        for score in (1,5):
            profile=typical.copy();profile[i]=(score-3)/2
            requests.append((RATINGS[i]+"_"+str(score),303,profile))
    strips=[]
    source_indices=list(dict.fromkeys([next((i for i,r in enumerate(data.rows) if r["totals"][name]>0),0)
                                      for name in ("civilians","wreck_mass")]))
    for i in source_indices:
        strips.append(layer_strip(data.targets[i].astype(np.float32),norm,"SOURCE training example "+data.rows[i]["key"]))
    metrics=[]
    for start in range(0,len(requests),batch_size):
        group=requests[start:start+batch_size]
        # Same initial noise for every setting of a given seed, independent of batch position.
        for name,seed,profile in group:
            ratings=torch.from_numpy(profile[None]).cuda()
            raw,bounded=sample(model,ratings,norm,seed,steps=48,guidance=1.5)
            physical=decode(raw[0],norm)
            np.savez_compressed(folder/(name+".npz"),raw_encoded=raw[0],sampler_bounded=bounded[0],
                physical=physical,ratings=(profile*2+3),seed=seed)
            strip=layer_strip(raw[0],norm,"GENERATED "+name+" | raw final estimate; no gameplay repair")
            strip.save(folder/(name+".png"))
            strips.append(strip)
            metrics.append(dict(name=name,seed=seed,ratings=(profile*2+3).tolist(),
                totals={CHANNELS[j]:float(physical[j].sum()) for j in range(2,10)},
                negative_quantity_fraction=float((physical[1:10]<-1e-5).mean()),
                water_fraction=float((physical[1]>.05).mean()),
                height_depth_sign_agreement=float(((physical[0]<0)==(physical[1]>.05)).mean())))
    combined=Image.new("RGB",(strips[0].width,sum(s.height for s in strips)))
    y=0
    for strip in strips:
        combined.paste(strip,(0,y));y+=strip.height
    combined.save(folder/"layers.png")
    shutil.copyfile(folder/"layers.png",Path(out)/"latest_layers.png")
    write_json(folder/"metrics.json",metrics)
    write_json(Path(out)/"latest_samples.json",dict(step=step,folder=str(folder),samples=metrics))
    return str(folder)


def setup(seed):
    if not torch.cuda.is_available():
        raise RuntimeError("CUDA required; refusing silent CPU training")
    torch.set_num_threads(4)
    torch.manual_seed(seed)
    torch.backends.cudnn.benchmark=False
    torch.set_float32_matmul_precision("high")
    torch.cuda.reset_peak_memory_stats()


def probe(args):
    setup(args.seed)
    model=JointUNet(args.base).cuda()
    ema=copy.deepcopy(model).eval().requires_grad_(False)
    optimizer=torch.optim.AdamW(model.parameters(),lr=1e-4)
    clean=torch.zeros(args.micro_batch,13,256,256,device="cuda")
    ratings=torch.zeros(args.micro_batch,12,device="cuda")
    cumulative=schedule(1000,"cuda")
    durations=[]
    for i in range(8):
        torch.cuda.synchronize();began=time.monotonic()
        optimizer.zero_grad(set_to_none=True)
        with torch.autocast("cuda",dtype=torch.bfloat16):
            losses,_=loss_terms(model,clean,ratings,cumulative,torch.Generator(device="cuda").manual_seed(i))
            loss=losses.mean()
        loss.backward()
        norm=nn.utils.clip_grad_norm_(model.parameters(),1.)
        if not torch.isfinite(loss) or not torch.isfinite(norm):
            raise ValueError("Nonfinite probe loss/gradient")
        optimizer.step()
        for a,b in zip(ema.parameters(),model.parameters(),strict=True):
            a.lerp_(b.detach(),.01)
        torch.cuda.synchronize();durations.append(time.monotonic()-began)
    result=dict(status="passed",base=args.base,micro_batch=args.micro_batch,
        parameters=sum(p.numel() for p in model.parameters()),
        peak_allocated_mb=torch.cuda.max_memory_allocated()/1024**2,
        peak_reserved_mb=torch.cuda.max_memory_reserved()/1024**2,
        seconds_per_microstep=float(np.median(durations[2:])),
        maps_per_second=args.micro_batch/float(np.median(durations[2:])))
    write_json(args.out,result)
    print(json.dumps(result),flush=True)


def train(args):
    setup(args.seed)
    out=args.out.resolve();out.mkdir(parents=True,exist_ok=True)
    with lease(out/"training.lock"):
        data=JointDataset(args.dataset,"train",args.overfit)
        validation=data if args.overfit else JointDataset(args.dataset,"validation")
        norm=data.norm
        model=JointUNet(args.base,dropout=0. if args.overfit else .1).cuda()
        ema=copy.deepcopy(model).eval().requires_grad_(False)
        optimizer=torch.optim.AdamW(model.parameters(),lr=args.learning_rate,weight_decay=0. if args.overfit else .01)
        cumulative=schedule(1000,"cuda")
        code=[Path(__file__),Path(__file__).with_name("multiscale_terrain.py"),
              Path(__file__).parents[1]/"tensors/joint_sources.py"]
        config=dict(schema="joint_source_diffusion_v1",dataset=str(args.dataset.resolve()),
            release_sha256=file_hash(args.dataset/"release_manifest.json"),base=args.base,
            target_channels=CHANNELS,targets=13,spatial_channels=0,profile_dim=12,ratings=RATINGS,
            micro_batch=args.micro_batch,effective_batch=args.batch_size,seed=args.seed,
            parameters=sum(p.numel() for p in model.parameters()),train_maps=len(data.rows),
            validation_maps=len(validation.rows),overfit=args.overfit,learning_rate=args.learning_rate,
            code={str(p):file_hash(p) for p in code},test_used=False,
            precision="bfloat16 compute; float32 parameters/optimizer; CPU source cache float16",
            objective="Equal channel mean of foreground-weighted velocity MSE; no geometry/placement auxiliary losses",
            sampling="DDIM 48 steps, CFG 1.5; training-range numerical bounds inside sampler; unbounded final estimate saved",
            schedule="500-step LR warmup then constant LR for resumable pilot; no pilot-end decay")
        old=read_json(out/"config.json") if (out/"config.json").exists() else None
        if old and old!=config:
            raise ValueError("Run contract differs; choose a new run")
        write_json(out/"config.json",config)
        write_json(out/"training_receipt.json",dict(config=config,dataset_audit=read_json(args.dataset/"audit.json"),
            generation_inputs=dict(noise=[13,256,256],timestep="scalar",ratings=[12],spatial_conditions=[]),
            supervised_targets=CHANNELS,source_contract=read_json(args.dataset/"contract.json"),
            masks="No input, target validity, spatial eligibility, or loss-exclusion masks",
            augmentation="Aligned full-stack rotations/reflections and start-channel exchange",
            postprocessing="No gameplay repair, symmetry projection, marker decoding, count caps or asset composition"))
        snapshot=out/"code_snapshot";snapshot.mkdir(exist_ok=True)
        for p in code:
            shutil.copyfile(p,snapshot/p.name)
        step,elapsed_before,best,history=0,0.,float("inf"),[]
        if args.resume:
            saved=torch.load(out/"last.pt",map_location="cuda",weights_only=False)
            if saved["config"]!=config:
                raise ValueError("Checkpoint contract mismatch")
            model.load_state_dict(saved["model"]);ema.load_state_dict(saved["ema"])
            optimizer.load_state_dict(saved["optimizer"])
            step,elapsed_before,best,history=(saved[k] for k in ("step","elapsed_seconds","best_validation","history"))
            torch.set_rng_state(saved["torch_rng"].cpu());torch.cuda.set_rng_state(saved["cuda_rng"].cpu())
            del saved
        elif (out/"last.pt").exists():
            raise ValueError("Existing checkpoint requires --resume")
        began=time.monotonic();recent=deque(maxlen=100)
        initial=validate(ema,validation,cumulative,args.micro_batch)
        if not (out/"initial_validation.json").exists():write_json(out/"initial_validation.json",initial)
        last_eval,last_gallery,last_save=0.,0.,0.
        print(json.dumps(dict(parameters=config["parameters"],initial=initial,train=len(data.rows),validation=len(validation.rows))),flush=True)

        def elapsed():return elapsed_before+time.monotonic()-began

        def progress(status,**extra):
            p=dict(status=status,step=step,elapsed_seconds=elapsed(),session_seconds=time.monotonic()-began,
                session_budget_seconds=args.seconds,parameters=config["parameters"],train_maps=len(data.rows),
                validation_maps=len(validation.rows),training_loss=float(np.mean(recent)) if recent else None,
                best_validation=best if math.isfinite(best) else None,
                peak_allocated_mb=torch.cuda.max_memory_allocated()/1024**2,
                peak_reserved_mb=torch.cuda.max_memory_reserved()/1024**2,
                maps_presented=step*args.batch_size,equivalent_epochs=step*args.batch_size/len(data.rows),
                seconds_per_update=(time.monotonic()-began)/max(1,step-start_step),updated_unix=time.time(),**extra)
            write_json(out/"progress.json",p)
            return p

        def save(metrics=None):
            nonlocal best
            improved=metrics is not None and metrics["validation_loss"]<best
            if metrics:
                best=min(best,metrics["validation_loss"])
                history.append(dict(step=step,elapsed_seconds=elapsed(),**metrics))
                write_json(out/"history.json",history)
            atomic_checkpoint(out/"last.pt",dict(model=model.state_dict(),ema=ema.state_dict(),
                optimizer=optimizer.state_dict(),config=config,normalization=norm,step=step,
                elapsed_seconds=elapsed(),best_validation=best,history=history,
                torch_rng=torch.get_rng_state(),cuda_rng=torch.cuda.get_rng_state()))
            if improved:
                atomic_checkpoint(out/"best.pt",dict(ema=ema.state_dict(),config=config,normalization=norm,
                    step=step,elapsed_seconds=elapsed(),metrics=metrics))

        start_step=step
        save(initial)
        reason="review_time_budget"
        while time.monotonic()-began<args.seconds:
            if (out/"pause.request").exists():reason="pause_requested";break
            if args.deadline and time.time()>args.deadline-120:break
            step+=1
            rng=np.random.default_rng(np.random.SeedSequence([args.seed,step]))
            generator=torch.Generator(device="cuda").manual_seed(args.seed*1000003+step)
            indices=rng.integers(len(data.rows),size=args.batch_size)
            optimizer.param_groups[0]["lr"]=args.learning_rate*min(1.,step/(50 if args.overfit else 500))
            model.train();optimizer.zero_grad(set_to_none=True)
            loss_sum=0.
            for offset in range(0,args.batch_size,args.micro_batch):
                clean,ratings=data.batch(indices[offset:offset+args.micro_batch],"cuda",None if args.overfit else rng)
                with torch.autocast("cuda",dtype=torch.bfloat16):
                    losses,_=loss_terms(model,clean,ratings,cumulative,generator)
                    loss=losses.mean()*len(clean)/args.batch_size
                if not torch.isfinite(loss):raise ValueError("Nonfinite joint loss")
                loss.backward();loss_sum+=float(loss.detach())
            grad=nn.utils.clip_grad_norm_(model.parameters(),1.)
            if not torch.isfinite(grad):raise ValueError("Nonfinite joint gradient")
            optimizer.step()
            with torch.no_grad():
                decay=min(.999,(1+step)/(10+step))
                for a,b in zip(ema.parameters(),model.parameters(),strict=True):a.lerp_(b.detach(),1-decay)
            recent.append(loss_sum)
            if step%10==0:progress("training")
            if step%50==0:print(f'step={step} loss={np.mean(recent):.5f} session={time.monotonic()-began:.0f}s',flush=True)
            since=time.monotonic()-began
            if since-last_save>=120:
                save();last_save=since
            if since-last_eval>=args.validate_seconds:
                metrics=validate(ema,validation,cumulative,args.micro_batch)
                save(metrics);last_eval=time.monotonic()-began
                print('validation='+json.dumps(metrics),flush=True)
            if not args.overfit and since-last_gallery>=args.gallery_seconds:
                progress("sampling")
                gallery(ema,data,out,step,norm,batch_size=args.micro_batch)
                last_gallery=time.monotonic()-began
        progress("finalizing")
        metrics=validate(ema,validation,cumulative,args.micro_batch);save(metrics)
        if not args.overfit:
            gallery(ema,data,out,step,norm,batch_size=args.micro_batch,full_sweep=False)
        else:
            initial_saved=read_json(out/"initial_validation.json")
            ratios={n:metrics["channel_velocity_loss"][n]/initial_saved["channel_velocity_loss"][n] for n in CHANNELS}
            passed=metrics["validation_loss"]<initial_saved["validation_loss"]*.8 and all(r<1 for r in ratios.values())
            write_json(out/"sanity_result.json",dict(status="passed" if passed else "failed",
                channel_loss_ratio=ratios,initial=initial_saved,final=metrics,
                meaning="Training-subset denoising implementation check; not generalization evidence"))
            if not passed:raise RuntimeError("Tiny-data learning check failed; inspect per-channel losses")
        result=progress("paused_for_review",reason=reason,**metrics)
        write_json(out/"summary.json",result)
        print(json.dumps(result),flush=True)


def main():
    parser=argparse.ArgumentParser(description=__doc__)
    parser.add_argument("mode",choices=("probe","train"))
    parser.add_argument("--dataset",type=Path,default=Path("data/tensors/joint_sources_curated_256_v1"))
    parser.add_argument("--out",type=Path,required=True)
    parser.add_argument("--base",type=int,default=128)
    parser.add_argument("--micro-batch",type=int,default=4)
    parser.add_argument("--batch-size",type=int,default=16)
    parser.add_argument("--seconds",type=float,default=2400)
    parser.add_argument("--deadline",type=float,default=0)
    parser.add_argument("--overfit",type=int)
    parser.add_argument("--learning-rate",type=float,default=2e-4)
    parser.add_argument("--seed",type=int,default=20261025)
    parser.add_argument("--validate-seconds",type=float,default=300)
    parser.add_argument("--gallery-seconds",type=float,default=600)
    parser.add_argument("--resume",action="store_true")
    args=parser.parse_args()
    if args.batch_size%args.micro_batch:raise ValueError("Microbatch must divide effective batch")
    if args.mode=="probe":probe(args)
    else:train(args)


if __name__=="__main__":main()
