"""Package the concluded joint-source experiment from saved artifacts only.

No checkpoint loading, inference, CUDA initialization, or training.
"""
from __future__ import annotations

import hashlib
import json
import os
import shutil
import statistics
from pathlib import Path

ROOT = Path(__file__).resolve().parents[1]
OUT = ROOT / "docs/model-atlas/research/all-in"
ASSETS = OUT / "assets"


def read(relative):
    return json.loads((ROOT / relative).read_text(encoding="utf-8"))


def main():
    ASSETS.mkdir(parents=True, exist_ok=True)
    sources = []
    rows = []
    specs = [
        ("16-map implementation check", "joint_sources_pilot_v1/sanity", 0, 24),
        ("Full-cohort pilot", "joint_sources_pilot_v1/pilot", 0, 24),
        ("Long continuation", "joint_sources_12h_v1/training", 2074, 24),
        ("One-map baseline", "joint_diagnostics_v1/one_map_baseline", 0, 6),
        ("One-map lower LR", "joint_diagnostics_v1/one_map_lower_lr", 0, 6),
        ("One-map balanced loss", "joint_diagnostics_v1/one_map_balanced", 0, 6),
        ("Full-cohort lower LR", "joint_diagnostics_v1/full_lower_lr", 2074, 24),
    ]
    for name, folder, start, batch in specs:
        relative = f"data/models/{folder}/summary.json"
        summary = read(relative)
        sources.append(relative)
        rows.append(dict(name=name, summary=relative, start_step=start,
                         end_step=summary["step"], additional_updates=summary["step"]-start,
                         effective_batch=batch,
                         map_presentations=(summary["step"]-start)*batch,
                         session_seconds=summary["session_seconds"]))
    telemetry = []
    pipelines = []
    for folder in ["joint_sources_pilot_v1", "joint_sources_12h_v1", "joint_diagnostics_v1"]:
        relative = f"data/models/{folder}/gpu_history.json"
        points = read(relative)
        sources.append(relative)
        pairs = [(a,b) for a,b in zip(points, points[1:]) if 0 < b["unix"]-a["unix"] < 120]
        telemetry.append(dict(run=folder, source=relative, samples=len(points),
            covered_seconds=sum(b["unix"]-a["unix"] for a,b in pairs),
            maximum_gap_seconds=max(b["unix"]-a["unix"] for a,b in pairs),
            mean_utilization_percent=statistics.mean(p["utilization_percent"] for p in points),
            median_utilization_percent=statistics.median(p["utilization_percent"] for p in points),
            peak_device_memory_mib=max(p["used_memory_mb"] for p in points),
            mean_board_power_watts=statistics.mean(p["power_watts"] for p in points),
            board_energy_estimate_kwh=sum((a["power_watts"]+b["power_watts"])/2 *
                (b["unix"]-a["unix"])/3600000 for a,b in pairs)))
        relative = f"data/models/{folder}/pipeline.json"
        pipeline = read(relative)
        sources.append(relative)
        pipelines.append(dict(run=folder, source=relative, elapsed_seconds=pipeline["elapsed_seconds"],
                              state=pipeline["state"]))
    copies = {
        "full-cohort-final.png": "data/models/joint_sources_12h_v1/training/latest_layers.png",
        "one-map-memorization.png": "data/models/joint_diagnostics_v1/one_map_baseline/latest_layers.png",
        "full-lower-lr.png": "data/models/joint_diagnostics_v1/full_lower_lr/latest_layers.png",
        "diagnostic-conclusions.png": "data/models/joint_diagnostics_v1/conclusions.png",
        "sampling-comparison.png": "data/models/joint_diagnostics_v1/sampling/comparison.png",
    }
    for name, relative in copies.items():
        shutil.copyfile(ROOT/relative, ASSETS/name)
        sources.append(relative)
    for relative in ["data/tensors/joint_sources_curated_256_v1/audit.json",
                     "data/tensors/joint_sources_curated_256_v1/release_manifest.json",
                     "data/tensors/joint_sources_curated_256_v1/normalization.json",
                     "data/models/joint_diagnostics_v1/comparison_summary.json",
                     "data/models/joint_diagnostics_v1/sampling/summary.json",
                     "data/models/joint_diagnostics_v1/objective_audit.json",
                     "data/models/joint_diagnostics_v1/oracle_precision.json",
                     "data/models/joint_sources_12h_v1/training/history.json",
                     "data/models/joint_diagnostics_v1/full_lower_lr/history.json"]:
        sources.append(relative)
    ledger = dict(schema="mapgen_all_in_research_compute_v1", research_status="closed_by_user",
        date="2026-10-25", device="One local NVIDIA GeForce RTX 5080, 16303 MiB, Windows WDDM",
        sessions=rows, session_seconds=sum(r["session_seconds"] for r in rows),
        map_presentations=sum(r["map_presentations"] for r in rows),
        supervisor_windows=pipelines, supervisor_seconds=sum(p["elapsed_seconds"] for p in pipelines),
        telemetry=telemetry,
        telemetry_board_energy_estimate_kwh=sum(t["board_energy_estimate_kwh"] for t in telemetry),
        accounting_notes=[
            "Session elapsed clocks include validation, galleries and checkpointing; not GPU kernel timers.",
            "Resumed cumulative elapsed and map counts are not added twice: start steps are subtracted.",
            "Supervisor windows overlap session clocks and must not be added to them.",
            "Separate GPU capacity/sampling probes and preparation are not fully timed; totals are measured core windows, not exhaustive project effort.",
            "Energy is trapezoidal integration of sampled GPU board power, excluding gaps >=120 seconds; not metered system electricity.",
            "No financial cost is inferred from local compute or power. No cloud training was purchased.",
        ],
        evidence=[dict(path=p, sha256=hashlib.sha256((ROOT/p).read_bytes()).hexdigest()) for p in sources])
    (OUT/"compute-ledger.json").write_text(json.dumps(ledger, indent=2)+"\n", encoding="utf-8")

    os.environ.setdefault("MPLCONFIGDIR", str(ROOT/".tmp/matplotlib-joint"))
    import matplotlib
    matplotlib.use("Agg")
    import matplotlib.pyplot as plt
    old = read("data/models/joint_sources_12h_v1/training/history.json")
    low = read("data/models/joint_diagnostics_v1/full_lower_lr/history.json")
    original = {p["step"]:p["validation_loss"] for p in old if "validation_loss" in p}
    lower = {p["step"]:p["metrics"]["500"]["loss"] for p in low}
    fig, axes = plt.subplots(1,2,figsize=(13,4.8),layout="constrained")
    for ax in axes:
        ax.plot(list(original),list(original.values()),"o-",color="#9a423b",ms=3,label="Original LR 2e-4")
        ax.plot(list(lower),list(lower.values()),"o-",color="#167366",ms=3,label="LR-only restart 5e-5")
        ax.set_xlabel("Cumulative optimizer updates")
        ax.set_ylabel("Held-out t=500 velocity loss (69 families)")
        ax.grid(alpha=.22)
    axes[0].set_yscale("log")
    axes[0].set_title("Full saved history: collapse, plateau, recovery")
    axes[0].legend(fontsize=9)
    axes[1].set(xlim=(2000,5100),ylim=(.021,.055),title="Controlled restart through the failure point")
    axes[1].axvline(4412,color="#999",ls="--",lw=1)
    axes[1].text(4412,.052,"Original collapse at 4,412",ha="right",fontsize=9)
    fig.suptitle("Lower denoising loss did not yield usable fresh maps",fontsize=15)
    fig.savefig(ASSETS/"learning-history.png",dpi=160)
    plt.close(fig)
    print(json.dumps({"session_seconds":ledger["session_seconds"],"session_hours":ledger["session_seconds"]/3600,
        "presentations":ledger["map_presentations"],"supervisor_seconds":ledger["supervisor_seconds"],
        "energy_estimate_kwh":ledger["telemetry_board_energy_estimate_kwh"]}))


if __name__ == "__main__":
    main()
