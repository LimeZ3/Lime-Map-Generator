def read(path):
    return json.loads(Path(path).read_text(encoding='utf-8-sig'))


def prepare():
    """Explicit local extraction; only official missing vault thumbnails use network."""
    from supmapai.tools.extract_layers_v2 import extract, load_resource_icons, vault_comparison
    from supmapai.tensors.map_textures import MaterialAssets, extract_texture_tensor
    CACHE.mkdir(parents=True, exist_ok=True)
    with (ROOT/'data/raw/faf_maps/latest_20261008/map_catalog.csv').open(encoding='utf-8-sig',newline='') as f:
        rows = {r['map_id']+'_'+r['version_id']:r for r in csv.DictReader(f)}
    bp, wp = read(PINNED_BP/'blueprint_index.json'), read(PINNED_BP/'wreck_value_profile.json')
    icons = load_resource_icons(GAME, CACHE)
    assets = MaterialAssets(GAME)
    manifest = []
    try:
        for key in KEYS:
            row, folder = rows[key], CACHE/key
            if not (folder/'record.json').exists():
                extract(row,CACHE,False,bp,icons,wp,include_settings=True)
            record = read(folder/'record.json')
            for path, sha in record['source_files'].items():
                if file_hash(Path(path)) != sha:
                    raise ValueError('Map source changed; rebuild into a new cache: '+key)
            if not (folder/'textures.npz').exists():
                scmap = next(Path(p) for p in record['source_files'] if p.endswith('.scmap'))
                arrays,meta,_ = extract_texture_tensor(scmap,assets,grid=256)
                np.savez_compressed(folder/'textures.npz',**arrays)
                (folder/'textures.json').write_text(json.dumps(meta,indent=2),encoding='utf-8')
            if not (folder/'vault_preview.png').exists():
                cached = next((p for p in (ROOT/'data/processed').glob('*/'+key+'/vault_preview.png') if p.parent!=folder),None)
                if cached:
                    shutil.copyfile(cached,folder/'vault_preview.png')
                elif vault_comparison(row,folder,True)['status'] != 'cached':
                    raise ValueError('Official vault preview unavailable: '+key)
            manifest.append({k:row[k] for k in ('map_id','version_id','display_name','folder_name','thumbnail_large','max_players','width','author_id','version_uploaded_at')})
            print('Prepared atlas source',key,flush=True)
    finally:
        assets.close()
    (CACHE/'maps.json').write_text(json.dumps(manifest,indent=2),encoding='utf-8')


def href(path):
    import os
    return Path(os.path.relpath(path,PAGE)).as_posix()


def raster(items, shape, value=None):
    result = np.zeros(shape,np.float32)
    for p in items:
        x,z = int(p['x']),int(p['z'])
        if 0<=z<shape[0] and 0<=x<shape[1]:
            result[z,x] += float(p.get(value) or 0) if value else 1
    return result


def rgb(values, low, high, diverging=False):
    t = np.clip((np.asarray(values,float)-low)/max(high-low,1e-12),0,1)
    palette = np.array([[16,24,33],[33,100,114],[82,170,134],[243,219,123]],float)
    if diverging:
        palette = np.array([[62,120,218],[233,238,235],[229,125,53]],float)
    s = t*(len(palette)-1);i = np.minimum(s.astype(int),len(palette)-2);f = (s-i)[...,None]
    return np.uint8(palette[i]*(1-f)+palette[i+1]*f)


def figure(path, pixels, title, legend, foot, points=(), extent=1024):
    font = ImageFont.truetype('C:/Windows/Fonts/segoeui.ttf',16)
    small = ImageFont.truetype('C:/Windows/Fonts/segoeui.ttf',13)
    image = Image.new('RGB',(512,600),'#ffffff')
    tile = Image.fromarray(np.uint8(pixels)).resize((512,512),Image.Resampling.NEAREST)
    image.paste(tile,(0,32));d = ImageDraw.Draw(image)
    d.text((10,6),title,fill='#20362a',font=font)
    for p, colour, label in points:
        x,z = p['x']/extent*512,p['z']/extent*512+32
        d.ellipse((x-4,z-4,x+4,z+4),fill=colour,outline='black')
        if label:d.text((x+5,z-8),label,fill='white',stroke_width=1,stroke_fill='black',font=small)
    d.text((10,550),legend,fill='#233b32',font=small)
    d.text((10,574),foot,fill='#5d6963',font=small)
    image.save(path)


def overlay_canvas():
    return Image.new('RGBA',(1024,1024),(0,0,0,0))


def draw_points(image, points, extent, color, *, labels=False, radius=6):
    draw=ImageDraw.Draw(image);font=ImageFont.truetype('C:/Windows/Fonts/segoeui.ttf',24)
    for point in points:
        x,z=point['x']/extent*1024,point['z']/extent*1024
        draw.ellipse((x-radius,z-radius,x+radius,z+radius),fill=color,outline='black',width=2)
        if labels:
            label=str(point.get('slot',point.get('name','').replace('ARMY_','')))
            draw.text((x+radius+3,z-14),label,font=font,fill='white',stroke_width=2,stroke_fill='black')


def mask_image(mask, color):
    rgba=np.zeros((*mask.shape,4),np.uint8);rgba[mask]=[*color,255]
    return Image.fromarray(rgba).resize((1024,1024),Image.Resampling.NEAREST)
